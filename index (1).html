return err && err.message ? err.message : String(err);
}
async function getJSON(url) {
var res = await fetch(url, { cache: "no-store" });
if (!res.ok) throw new Error("Server replied with HTTP " + res.status + ".");
var data = await res.json();
if (data && data.error) throw new Error(String(data.error));
return data;
}

function dueInfo(due) {
var blank = due === null || due === undefined || due === "";
var n = Number(due);
if (blank) return { text: "–", unit: "", now: false };
if (Number.isNaN(n)) return { text: String(due), unit: "", now: false };
if (n === 0) return { text: "Due", unit: "", now: true };
return { text: String(n), unit: "min", now: false };
}

if (!trams.length) {

/* ---------- Refresh ---------- */
async function refresh() {
var mine = ++requestCounter;
var errors = [];
if (!stopsLoaded) {
try { await loadStops(); }
catch (e) {
errors.push("Couldn’t load the list of stops: " + describeError(e));
if (!selectEl.options.length || !stops.length) fallbackSelect();
}
}
var stopAtRequest = currentId;
try {
var data = await getJSON(BASE + "/forecast?stop=" + encodeURIComponent(stopAtRequest));
if (mine !== requestCounter) return; // a newer request took over
if (!data || !Array.isArray(data.directions)) {
throw new Error("The forecast came back in an unexpected format.");
}
render(data);
lastUpdated = new Date();
updateStatus();
} catch (e) {
if (mine !== requestCounter) return;
errors.push("Couldn’t load live trams: " + describeError(e));
if (!resultsEl.querySelector(".card, .board-wrap")) {
resultsEl.textContent = "";
stopNameEl.textContent = "Stop " + stopAtRequest;
}
}
showErrors(errors);
}

noticeEl.hidden = true;
resultsEl.textContent = "";
resultsEl.appendChild(el("div", "loading", "Loading trams…"));
lastUpdated = null;
lastData = null;
updateStatus();
refresh();
}
/* ---------- Nearest stop (geolocation + haversine) ---------- */
function toRad(deg) { return deg * Math.PI / 180; }
// Great-circle distance between two lat/lng points, in kilometres
function haversineKm(lat1, lng1, lat2, lng2) {
var R = 6371;
var dLat = toRad(lat2 - lat1);
var dLng = toRad(lng2 - lng1);
var a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
Math.sin(dLng / 2) * Math.sin(dLng / 2);
return 2 * R * Math.asin(Math.min(1, Math.sqrt(a)));
}
function formatDistance(km) {
var m = km * 1000;
if (m < 100) return Math.round(m) + " m";
if (m < 1000) return (Math.round(m / 10) * 10) + " m";
if (km < 100) return km.toFixed(1) + " km";
return Math.round(km) + " km";
}
function formatWalk(km) {
var minutes = km / WALK_KMH * 60;
if (minutes < 1) return "under 1 min walk";
var m = Math.round(minutes);
if (m < 60) return "about " + m + " min walk";
var h = Math.floor(m / 60);
var r = m % 60;
return "about " + h + " h" + (r ? " " + r + " min" : "") + " walk";
}
function getPosition() {
return new Promise(function (resolve, reject) {
navigator.geolocation.getCurrentPosition(resolve, reject, {
enableHighAccuracy: true,
timeout: 10000,
maximumAge: 60000
});
});
}
function geoErrorMessage(err) {
var secureHint = window.isSecureContext ? "" : " This page isn’t served over https, which browsers require for location.";
if (err && typeof err.code === "number") {
if (err.code === 1) return "Location permission was denied. Allow location access for this page in your browser settings and try again." + secureHint;
if (err.code === 2) return "Your location couldn’t be determined. Check that location services are switched on." + secureHint;
if (err.code === 3) return "Finding your location took too long. Please try again.";
}
return describeError(err);
}
function showNearest(stop, km) {
nearInfoEl.textContent = "";
var title = el("div", null, "Nearest stop: ");
title.appendChild(el("strong", null, stop.name));
nearInfoEl.appendChild(title);
nearInfoEl.appendChild(el("div", null, formatDistance(km) + " away · " + formatWalk(km)));
var note = "Straight-line distance, walking at " + WALK_KMH + " km/h.";
if (km > 20) note += " You seem to be a long way from the Luas network.";
nearInfoEl.appendChild(el("span", "near-small", note));
nearInfoEl.hidden = false;
}
async function findNearest() {
nearErrorEl.hidden = true;
nearBtn.disabled = true;
nearLabel.textContent = "Locating…";
try {
if (!("geolocation" in navigator)) {
throw new Error("This browser doesn’t support location.");
}
if (!stopsLoaded) {
try { await loadStops(); }
catch (e) { throw new Error("Couldn’t load the list of stops: " + describeError(e)); }
}
var pos = await getPosition();
var lat = pos.coords.latitude;
var lng = pos.coords.longitude;
var best = null;
stops.forEach(function (s) {
var sLat = Number(s.lat), sLng = Number(s.lng);
if (!isFinite(sLat) || !isFinite(sLng)) return;
var d = haversineKm(lat, lng, sLat, sLng);
if (!best || d < best.km) best = { stop: s, km: d };
});
if (!best) throw new Error("None of the stops have usable coordinates.");
showNearest(best.stop, best.km);
selectStop(best.stop.id);
} catch (e) {
nearInfoEl.hidden = true;
nearErrorEl.textContent = e && e.code ? geoErrorMessage(e) : describeError(e);
nearErrorEl.hidden = false;
} finally {
nearBtn.disabled = false;
nearLabel.textContent = "Nearest stop";
}
}
